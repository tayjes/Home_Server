sudo ./venv/bin/uvicorn lib.main:app --host 0.0.0.0 --port 8000 --reload

source venv/bin/activate

#to build the container
sudo docker build -t network-scanner .

#run the container you need to add some flag for permission
sudo docker run \
  --network host \
  --cap-add NET_RAW \
  --cap-add NET_ADMIN \
  -e MAC_DB_PATH=/app/helper/MAC.txt \
  network-scanner
